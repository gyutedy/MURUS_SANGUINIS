#!/usr/bin/env python3
"""elocution.py — 2e temps de SF_06_LIBELLUS. La méthode écrit la trame
(titres.py, 12 gabarits), la grâce polit les mots (API NVIDIA NIM),
le validateur de règles re-juge TOUTE sortie. Un titre que le gabarit
refuse est rejeté. Le verdict de sélection (M1) est déjà FERME :
ce module ne peut pas influer dessus — il tourne APRÈS M1."""
import json, os, time, urllib.request
from pathlib import Path

NIM_URL = "https://integrate.api.nvidia.com/v1/chat/completions"  # OpenAI-compatible
MODEL   = os.environ.get("MURUS_ELOCUTION_MODEL", "meta/llama-3.2-3b-instruct")

# Règles copywriting non-averti [source : CONTRACTS/copywriting_doctrine.md,
# doctrine existante du dépôt — le prompt n'invente pas les règles, il les cite]
PROMPT = """Reecris ce titre de clip pour quelqu'un qui ne connait PAS le streamer.
Regles non negociables :
1. Aucun nom propre (ni streamer, ni jeu) dans le hook.
2. Une question ouverte ou une promesse concrete en premiere ligne.
3. Un chiffre concret plutot qu'un adjectif.
4. La promesse est payee avant la 3e seconde de lecture.
5. Ne change PAS l'ordre d'information du titre (structure = loi).
6. Maximum 60 caracteres. Retourne UNIQUEMENT le titre reecrit.
Titre a reecrire :
"""

def elocuter(titre: str, retries: int = 3) -> dict:
    cle = os.environ["NVIDIA_API_KEY"]           # secret GitHub, jamais en clair
    corps = json.dumps({"model": MODEL, "messages": [
        {"role": "user", "content": PROMPT + titre}]}).encode()
    for i in range(retries):
        try:
            req = urllib.request.Request(NIM_URL, data=corps, headers={
                "Authorization": f"Bearer {cle}", "Content-Type": "application/json"})
            with urllib.request.urlopen(req, timeout=30) as r:
                candidat = json.loads(r.read())["choices"][0]["message"]["content"].strip()
            if valide(candidat):                 # LE GABARIT RE-JUGE — la loi tient
                return {"titre": candidat, "origine": "elocution", "essais": i + 1}
        except Exception as e:
            time.sleep([60, 300, 1500][i])       # backoff 1 min / 5 min / 25 min
            if i == retries - 1: dernier = str(e)
    # Repli gabarit pur : un siege n'echoue JAMAIS pour un titre
    return {"titre": titre, "origine": "gabarit", "elocution_fallback": True,
            "raison": dernier if 'dernier' in dir() else "refuse par le validateur"}

def valide(t: str) -> bool:
    return 0 < len(t) <= 60 and t != PROMPT.strip() and "\n" not in t
# Le validateur COMPLET est celui de titres.py (les 12 gabarits) : il est
# appele par le corps du module M2 reel — ici, seule la frontiere est montree.

if __name__ == "__main__":
    import sys
    pack = json.loads(Path(sys.argv[1]).read_text(encoding="utf-8"))
    for a in pack["assets"]:
        a["metadata"]["overlay_title"] = elocuter(a["metadata"]["overlay_title"])
    Path(sys.argv[2]).write_text(json.dumps(pack, indent=2), encoding="utf-8")

def attendre(url: str, essais: int = 60) -> None:

Philosophie — le mur, la rampe, la voix, l'heure

Le mur est la méthode : quatorze portes, chaque brèche validée avant d'y engager la suivante — l'héritage Iron Warriors, la guerre de siège contre les forteresses-plateformes. La rampe est la vitesse : le live ne pardonne pas l'attente, et le moteur greffé (volé à l'ennemi, sous licence MIT) frappe vite — mais derrière les mêmes portes, et il ne touchera jamais aux métadonnées : la patience du camouflage reste la nôtre. La voix est la grâce : la méthode écrit la trame, Sanguinius polit les mots — le modèle NVIDIA ne choisit jamais ce qui sera dit, il dit plus humainement ce que les règles ont déjà décidé, et ce qu'il dit repasse devant le juge qui a fait le mur. L'heure est la patience organisée : un siège qui frappe à intervalle fixe apprend à la forteresse à parer ; l'intervalle 4–10 h tiré une fois, signé, auditable, est imprévisible pour l'ennemi et vérifiable pour le ledger. Le cerveau réfléchit, le bras frappe, la voix séduit, le calendrier choisit l'heure — et le ledger juge tout, y compris le hasard.

La traçabilité comme seule valeur reste la loi des lois : un dépôt « incroyable » non vérifié vaut zéro ; un moteur modeste, cloné, exécuté et mesuré vaut de l'argent. Ce document ne cite que ce qui a été ouvert — chaque fichier du tree/F trace vers son origine, chaque test porte son oracle, chaque chiffre non mesuré porte ATTENDU.

Réflexion — pourquoi CETTE architecture

  • Pourquoi deux moteurs et pas un : le VOD et le live ont des exigences opposées (grâce vs vitesse) et les deux moteurs existent, testés chacun dans son domaine. Les fusionner en un seul serait réécrire l'un ou affaiblir l'autre. Les faire consommer le même pack, derrière la même porte, les met sous la même loi — c'est moins cher qu'une réécriture et plus sûr qu'un outil parallèle non gouverné.
  • Pourquoi le moteur greffé est vendor et pas fork : un fork vit de l'amont ; un vendor figé fonctionne sans lui. La survie des forges (R18) impose l'indépendance : NeonClipping peut mourir demain, la voie vive continue de rendre.
  • Pourquoi l'élocution est dans SF_06 et pas une sous-frégate : parce qu'elle ne produit pas d'artefact — elle transforme celui de la sous-frégate. Une sous-frégate sans artefact propre serait une porte en trop, ce que la loi interdit (une gate qui n'ouvre pas de sous-frégate est une clé morte).
  • Pourquoi SF_17 est la dernière du chapitre : parce qu'elle consomme tout ce qui la précède — les clips validés de V7 et le calendrier validé de V8. Publier avant d'avoir des clips conformes serait confondre la guerre avec sa propagande.
  • Ce qui reste incertain NON VÉRIFIÉ : le temps réel par clip live sur runner sans GPU ; la consommation réelle de crédits NVIDIA ; le scheduling Reels d'Instagram. Tout est mesuré au premier siège et écrit ici — rien n'est tranché par la réflexion, seulement par le monde.
